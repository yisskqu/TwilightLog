核心实现 · 18 行 
// srcset 的每一档都走同一条路径
pub fn derive(src: &Path, widths: &[u32]) -> Vec<Output> {
    widths
        .par_iter()
        .map(|&w| {
            let img = decode(src)?;
            let out = resize(&img, w);
            encode(out, w)
        })
        .collect()
}
